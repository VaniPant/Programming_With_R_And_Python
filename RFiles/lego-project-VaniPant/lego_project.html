1. Aim

The project applies concepts from R and tidyverse lectures to analyze a dataset of official Lego sets from rebrickable.com. It focuses on creating a structured Quarto document and implementing the necessary infrastructure for effective data exploration and presentation.

4.5. Sampling Larger Number of Rows

sample_sizes <- c(10000, 20000, 50000, 100000)

results <- list()

for (n in sample_sizes) {
  set.seed(123)
  sampled_inventory_parts <- inventory_parts |>
    slice_sample(n = n, replace = TRUE)
  
  sampled_inventory_ids <- sampled_inventory_parts |>
    distinct(inventory_id)
  
  time_filter <- system.time({
    filtered_parts_filter <- inventory_parts |>
      filter(inventory_id %in% sampled_inventory_ids$inventory_id)
  })
  
  time_semi_join <- system.time({
    filtered_parts_semi_join <- inventory_parts |>
      semi_join(sampled_inventory_ids, by = "inventory_id")
  })
  
  time_inner_join <- system.time({
    filtered_parts_inner_join <- inventory_parts |>
      inner_join(sampled_inventory_ids, by = "inventory_id")
  })
  
  results[[as.character(n)]] <- data.frame(
    sample_size = n,
    method = c("filter", "semi_join", "inner_join"),
    time_user = c(time_filter["user.self"], time_semi_join["user.self"], time_inner_join["user.self"]),
    time_sys = c(time_filter["sys.self"], time_semi_join["sys.self"], time_inner_join["sys.self"]),
    time_elapsed = c(time_filter["elapsed"], time_semi_join["elapsed"], time_inner_join["elapsed"])
  )
}

benchmark_results_extended <- do.call(rbind, results)
